# gd01 for PS Vita: builds the upstream gd-game-01 sources (read-only, from
# ../upstream) plus the Vita layer in this folder. Configure with the VitaSDK
# toolchain; tools/build-vita.sh does it all.
cmake_minimum_required(VERSION 3.24)

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
  else()
    message(FATAL_ERROR "Please define VITASDK to point to your SDK path!")
  endif()
endif()

project(gd01_vita VERSION 1.1.11 LANGUAGES C CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)

set(VITA_APP_NAME "gd01")
set(VITA_TITLEID  "GDDE00001")   # placeholder title id, 9 chars
set(VITA_VERSION  "01.00")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(UPSTREAM  "${CMAKE_CURRENT_SOURCE_DIR}/../upstream" CACHE PATH "prj-gd-engine checkout")
set(GD01_SRC  "${UPSTREAM}/gd-game-01/engine/src")
set(GDCORE    "${UPSTREAM}/gd-engine-core")
# SPIR-V headers generated by the desktop build (host glslang). They are
# platform-independent data; tools/build-vita.sh copies them here.
set(GD01_GEN  "${CMAKE_BINARY_DIR}/gen" CACHE PATH "generated headers")

find_package(SDL3 REQUIRED CONFIG)

# --- gd-engine-core, reduced: OpenGL render backend only, no audio backends ---
add_library(gdcore_render STATIC
  ${GDCORE}/render/src/device.cpp
  ${GDCORE}/render/src/common/spirv_reflect.cpp
  ${GDCORE}/render/src/gl/gl_api.cpp
  ${GDCORE}/render/src/gl/gl_device.cpp
)
target_include_directories(gdcore_render PUBLIC ${GDCORE}/render/include PRIVATE ${GDCORE}/render/src)
target_compile_definitions(gdcore_render PRIVATE GDCORE_RENDER_OPENGL3=1 GDCORE_RENDER_SDLGPU=0 GDCORE_RENDER_VULKAN=0)
target_link_libraries(gdcore_render PRIVATE SDL3::SDL3)

add_library(gdcore_audio STATIC ${GDCORE}/audio/src/output.cpp)
target_include_directories(gdcore_audio PUBLIC ${GDCORE}/audio/include)
target_include_directories(gdcore_audio PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/compat)  # dlfcn.h shim
target_compile_definitions(gdcore_audio PUBLIC GDCORE_HAS_AUDIO=1)
target_link_libraries(gdcore_audio PRIVATE SDL3::SDL3)

# --- gd01 ---

list(TRANSFORM GD01_SOURCES PREPEND "${GD01_SRC}/")

add_executable(gd01 ${GD01_SOURCES} src/vita_platform.cpp src/glsl_es100.cpp src/texture_palette.cpp src/movie_worker.cpp src/mpeg2_neon.cpp)
# SDL calls the Vita layer (src/vita_platform.cpp) takes over, so the
# upstream sources stay unmodified.
foreach(sym SDL_CreateWindow SDL_GL_CreateContext SDL_GL_DestroyContext SDL_GL_MakeCurrent
            SDL_GL_SwapWindow SDL_GL_SetSwapInterval SDL_GL_GetProcAddress SDL_GL_ExtensionSupported
            SDL_GetBasePath SDL_GetPrefPath)
  target_link_options(gd01 PRIVATE "-Wl,--wrap=${sym}")
endforeach()
# vita/src is on the path for the patches' Vita headers (e.g. movie_worker.h in patch 0006).
target_include_directories(gd01 PRIVATE ${GD01_SRC} ${GD01_GEN} ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(gd01 PRIVATE GD01_DLSS=0 GD01_VITA=1 GD01_VERSION="${PROJECT_VERSION}")
target_link_libraries(gd01 PRIVATE
  gdcore_render gdcore_audio
  SDL3::SDL3
  vitaGL vitashark SceShaccCgExt mathneon taihen_stub kubridge_stub
  SceShaccCg_stub SceGxm_stub SceDisplay_stub SceSysmodule_stub SceKernelDmacMgr_stub
  SceAppMgr_stub SceAppUtil_stub SceCommonDialog_stub SceIme_stub
  SceCtrl_stub SceTouch_stub SceMotion_stub SceHid_stub ScePower_stub
  SceAudio_stub SceAudioIn_stub
  z m stdc++
)

# The Guitar Dream icon PNGs built into the program (same step as upstream).
set(GD01_ENGINE "${UPSTREAM}/gd-game-01/engine")
set(GD01_ICON_FILES gd01_icon gd01_icon_small)
set(GD01_ICON_SYMBOLS kGd01IconPng kGd01IconSmallPng)
foreach(file symbol IN ZIP_LISTS GD01_ICON_FILES GD01_ICON_SYMBOLS)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${file}_data.cpp
    COMMAND ${CMAKE_COMMAND} -DIN=${GD01_ENGINE}/res/${file}.png -DOUT=${CMAKE_CURRENT_BINARY_DIR}/${file}_data.cpp
            -DNAME=${symbol} -P ${GD01_ENGINE}/cmake/embed_file.cmake
    DEPENDS ${GD01_ENGINE}/res/${file}.png ${GD01_ENGINE}/cmake/embed_file.cmake
    VERBATIM)
  target_sources(gd01 PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${file}_data.cpp)
endforeach()

# Extended memory mode (ATTRIBUTE2=12): about 100 MB more main RAM for the
# app, the standard setting of large Vita homebrew ports. The quick play
# character preview ran out of the default budget (hardware, 2026-11-10).
set(VITA_MKSFOEX_FLAGS "${VITA_MKSFOEX_FLAGS} -d ATTRIBUTE2=12")
# Allocation failures (src/vita_platform.cpp logs them with their size).
# One-shot arguments from ux0:data/gd01/args.txt (src/vita_platform.cpp).
target_link_options(gd01 PRIVATE "-Wl,--wrap=main")
foreach(sym malloc calloc realloc memalign)
  target_link_options(gd01 PRIVATE "-Wl,--wrap=${sym}")
endforeach()
# vita-elf-create appends ~4.5 KB of SCE tables after the code segment; when
# the code ends just under a 64 KiB boundary the data segment overlaps them
# ("Cannot allocate ... segment 1 overlaps"). The SDK's linker script reserves
# __sce_headroom bytes before the data segment.
target_link_options(gd01 PRIVATE "-Wl,--defsym=__sce_headroom=0x2000")

vita_create_self(eboot.bin gd01 UNSAFE)
vita_create_vpk(gd01.vpk ${VITA_TITLEID} eboot.bin
  VERSION ${VITA_VERSION}
  NAME ${VITA_APP_NAME}
)
